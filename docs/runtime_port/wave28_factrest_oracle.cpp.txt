// WAVE 28 lane FACTREST — OFFLINE BYTE ORACLE for the 35 non-UI factory pairs.
//
// ADAPTED FROM lane FACTUI1's oracle (docs/runtime_port/wave28_factui1_oracle.cpp.txt).
// The method is theirs; what changed is that this one includes THIS LANE'S OWN
// COMMITTED TABLES verbatim (runtime/src/<ns>/rt_factories_<ns>.inc) rather than
// a hand-copy of them, so the thing under test is the thing that shipped.
//
// WHAT IT COMPARES, AND WHAT THAT IS WORTH PER SHAPE — read this before the score
// -----------------------------------------------------------------------------
// Shape (b), the 13 classes with a fully inlined constructor: OUR transcribed
//   step list runs on buffer A, the DLL's OWN ?CreateInPlaceObject@ runs on
//   buffer B, and the bytes either side wrote are compared. THIS IS A REAL
//   ORACLE — our store list is being checked against retail's stores.
//
// Shape (c), the 14 delegating classes: our body calls the DLL's ??0C@NS@@ and
//   so does theirs, so the ctor's own stores CANNOT disagree. What is tested is
//   only the WRAPPER — that we call the same constructor, add no store and drop
//   none. Scored SEPARATELY below and labelled WRAPPER-ONLY. Counting these in
//   with the real comparisons would be the shadow-marker defect
//   (reference_shadow_marker_is_not_an_oracle.md): a check whose two sides share
//   the body they are supposed to be comparing.
//
// Shape (a), the 10 abstract classes: both sides must return nullptr. Trivial,
//   but it is the whole of retail's behaviour for those 20 slots, and the
//   negative control below can still be armed on it.
//
// EXCLUDED FROM THE BYTE COMPARISON, with the reason:
//   * +0x08..+0x0f of any object whose ctor chain reaches ??0Internal@Object@@
//     (AchievementInstance, GoalInstance). That ctor calls ?Rand@Random@System@@
//     and InternalFactory hands the SECOND caller a different handle id BY
//     DESIGN. Comparing it would report a divergence that is correct behaviour.
//   * Nothing else. The vptr IS compared.
//
// EFFECTIVENESS CONTROL: --negative corrupts one store value per class before
// the comparison; every checked class must then FAIL. A screen that cannot be
// made to fail is not a screen. Per FACTUI1's finding, a class with no
// immediate store falls back to corrupting a vtable value, so no class is
// silently left unarmed.

#include <windows.h>
#include <cstdio>
#include <cstdint>
#include <cstring>
#include <cstdlib>

// ---- THIS LANE'S COMMITTED CONTRACT, copied verbatim from ------------------
// runtime/src/object/rt_factories.h. A field-order mismatch here would be a
// silent misparse of the .inc, not a compile error, so it is copied whole.
enum class RtFactOp : uint8_t { End = 0, Vptr, Imm, Self, Call };

struct RtCtorStep {
    RtFactOp    kind;
    uint8_t     size;
    uint16_t    off;
    uint64_t    imm;
    const char* name;
    uint64_t    va;
};

struct RtFactoryClass {
    const char*       typeName;
    uint32_t          allocSize;
    uint32_t          allocAlign;
    const RtCtorStep* newSteps;
    const RtCtorStep* inPlaceSteps;
    uint64_t          newVa;
    uint64_t          inPlaceVa;
    uint64_t*         counters;
};

struct RtFactoryReg {
    const char*           newName;
    uint64_t              newVa;
    void*                 newFn;
    const char*           inPlaceName;
    uint64_t              inPlaceVa;
    void*                 inPlaceFn;
    const RtFactoryClass* cls;
};

// The .inc files' extern "C" wrappers call these two; the oracle drives the
// step lists directly, so these only need to link.
static void* RtFactoryNew(const RtFactoryClass&) { return nullptr; }
static void* RtFactoryInPlace(const RtFactoryClass&, void* p) { return p; }

// Each .inc ends with its own `static const RtFactoryReg kRegs[]`, so each gets
// its own namespace. Nothing else in them collides.
#include "inc/rt_factories_object.inc.h"
#include "inc/rt_factories_graphics.inc.h"
#include "inc/rt_factories_video.inc.h"
#include "inc/rt_factories_achievement.inc.h"
#include "inc/rt_factories_collision2d.inc.h"
#include "inc/rt_factories_audio.inc.h"
#include "inc/rt_factories_io.inc.h"

struct Group { const char* name; const RtFactoryReg* regs; int n; };
static const Group kGroups[] = {
    { "Object",       f_object::kRegs,       (int)(sizeof(f_object::kRegs) / sizeof(RtFactoryReg)) },
    { "Graphics",     f_graphics::kRegs,     (int)(sizeof(f_graphics::kRegs) / sizeof(RtFactoryReg)) },
    { "Video",        f_video::kRegs,        (int)(sizeof(f_video::kRegs) / sizeof(RtFactoryReg)) },
    { "Achievement",  f_achievement::kRegs,  (int)(sizeof(f_achievement::kRegs) / sizeof(RtFactoryReg)) },
    { "Collision2D",  f_collision2d::kRegs,  (int)(sizeof(f_collision2d::kRegs) / sizeof(RtFactoryReg)) },
    { "Audio",        f_audio::kRegs,        (int)(sizeof(f_audio::kRegs) / sizeof(RtFactoryReg)) },
    { "IO",           f_io::kRegs,           (int)(sizeof(f_io::kRegs) / sizeof(RtFactoryReg)) },
    { "Pathfinding",  f_pathfinding::kRegs,  (int)(sizeof(f_pathfinding::kRegs) / sizeof(RtFactoryReg)) },
    { "Picking",      f_picking::kRegs,      (int)(sizeof(f_picking::kRegs) / sizeof(RtFactoryReg)) },
};

static HMODULE g_dll;
static void* Sym(const char* n) { return (void*)GetProcAddress(g_dll, n); }
typedef void* (*CtorFn)(void*);

static int StepCount(const RtCtorStep* s) {
    int n = 0;
    if (!s) return 0;
    while (s[n].kind != RtFactOp::End) ++n;
    return n;
}

static bool HasCall(const RtCtorStep* s) {
    for (int i = 0; s && s[i].kind != RtFactOp::End; ++i)
        if (s[i].kind == RtFactOp::Call) return true;
    return false;
}

static bool ReachesInternal(const RtCtorStep* s) {
    for (int i = 0; s && s[i].kind != RtFactOp::End; ++i)
        if (s[i].kind == RtFactOp::Call && strstr(s[i].name, "Internal@Object@@")) return true;
    return false;
}

// OUR CreateInPlaceObject, exactly as rt_factories_core.cpp runs it.
static bool OursInPlace(const RtCtorStep* steps, void* self, const char* cls) {
    unsigned char* p = (unsigned char*)self;
    for (int i = 0; steps[i].kind != RtFactOp::End; ++i) {
        const RtCtorStep& s = steps[i];
        if (s.kind == RtFactOp::Vptr) {
            // The ??_7 export's ADDRESS is the vtable. Do NOT dereference it:
            // that yields vtable[0], the first virtual function, and stamping
            // that produced a 3-byte divergence at +0x00 on all eleven shape-(b)
            // classes -- an instrument defect that read exactly like a
            // catastrophic port. rt_factories_core.cpp stores ResolveExport()'s
            // return value, which is this address, so this now matches the port.
            void* vt = Sym(s.name);
            memcpy(p + s.off, &vt, 8);
        } else if (s.kind == RtFactOp::Imm) {
            uint64_t v = s.imm;
            memcpy(p + s.off, &v, s.size);
        } else if (s.kind == RtFactOp::Self) {
            void* q = p + s.imm;
            memcpy(p + s.off, &q, 8);
        } else if (s.kind == RtFactOp::Call) {
            CtorFn f = (CtorFn)Sym(s.name);
            if (!f) { printf("  !! %s: ctor %s NOT FOUND\n", cls, s.name); return false; }
            f(self);
        }
    }
    return true;
}

int main(int argc, char** argv) {
    setvbuf(stdout, nullptr, _IONBF, 0);
    const bool neg = (argc > 1 && strcmp(argv[1], "--negative") == 0);
    const bool runRisky = getenv("RISKY") != nullptr;
    g_dll = LoadLibraryA("C:\\GOG Games\\Banished\\Runtime-x64.dll");
    if (!g_dll) { printf("LoadLibrary failed %lu\n", GetLastError()); return 2; }

    void** allocSlot = (void**)Sym("?_singleton@MemoryAllocator@System@@0PEAV12@EA");
    void** ifSlot    = (void**)Sym("?_singleton@InternalFactory@Object@@0PEAV12@EA");

    if (allocSlot && !*allocSlot) {
        void* store = calloc(1, 1 << 16);
        typedef void* (*C0)(void*); typedef bool (*S2)(void*, size_t, size_t);
        C0 c = (C0)Sym("??0MemoryAllocator@System@@QEAA@XZ");
        S2 st = (S2)Sym("?Startup@MemoryAllocator@System@@QEAA_N_K0@Z");
        // NOTE (FACTUI1): Startup's two arguments are in MEGABYTES -- it shifts
        // them left by 20 -- and arg1 is a maximum that must be >= arg0, which
        // is why passing byte counts makes it silently return false.
        if (c && st) {
            c(store);
            size_t a0 = getenv("A0") ? strtoull(getenv("A0"), 0, 0) : 8;
            size_t a1 = getenv("A1") ? strtoull(getenv("A1"), 0, 0) : 256;
            printf("MemoryAllocator ctor+Startup(%llu,%llu) -> %d, singleton %p\n",
                   (unsigned long long)a0, (unsigned long long)a1,
                   (int)st(store, a0, a1), *allocSlot);
        }
    }
    if (ifSlot && !*ifSlot && allocSlot && *allocSlot) {
        void* store = calloc(1, 1 << 16);
        typedef void* (*C0)(void*); typedef bool (*S0)(void*);
        C0 c = (C0)Sym("??0InternalFactory@Object@@QEAA@XZ");
        S0 st = (S0)Sym("?Startup@InternalFactory@Object@@QEAA_NXZ");
        if (c && st) { c(store); printf("InternalFactory ctor+Startup -> %d, singleton %p\n",
                                        (int)st(store), *ifSlot); }
    }

    int realChk = 0, realPass = 0;       // shape (b): a genuine store comparison
    int wrapChk = 0, wrapPass = 0;       // shape (c): WRAPPER ONLY
    int stubChk = 0, stubPass = 0;       // shape (a): both must return nullptr
    int skipped = 0;

    for (const Group& g : kGroups) {
        for (int i = 0; i < g.n; ++i) {
            const RtFactoryReg& r = g.regs[i];
            const RtFactoryClass& c = *r.cls;
            const char* only = getenv("ONLY");
            if (only && !strstr(c.typeName, only)) continue;
            CtorFn theirs = (CtorFn)Sym(r.inPlaceName);
            if (!theirs) { printf("%-28s SKIP (export missing)\n", c.typeName); ++skipped; continue; }

            // ---- shape (a): both sides must return null --------------------
            if (!c.inPlaceSteps) {
                static unsigned char z[0x400];
                memset(z, 0xA5, sizeof(z));
                void* t = theirs(z);
                bool ours_null = true;           // our body returns nullptr by construction
                if (neg) ours_null = false;      // the control: claim it returns the buffer
                ++stubChk;
                bool ok = (t == nullptr) && ours_null;
                if (ok) ++stubPass;
                printf("%-28s %s  (abstract: both return null)\n", c.typeName,
                       ok ? "OK  " : "FAIL");
                continue;
            }

            const bool deleg  = HasCall(c.inPlaceSteps);
            const bool handle = ReachesInternal(c.inPlaceSteps);

            // The five Video classes and EdgeMesh/FrameCopyImageBuffer reach
            // ?PlatformConstruct@ on the video Device, which this bare
            // LoadLibrary process has not stood up. Guarded behind RISKY=1 so a
            // crash cannot be mistaken for a verdict, and a SKIP is never read
            // as a pass (FACTUI1's warning 3).
            static const char* kRisky[] = {
                "Graphics::EdgeMesh", "Graphics::FrameCopyImageBuffer",
                "Video::ImageBuffer", "Graphics::ReflectionImageBuffer",
                "Video::PixelProgram", "Video::RenderState", "Video::VertexLayout",
                "Video::VertexProgram", "Video::VideoProgram", "IO::PackageFile",
                "Graphics::ParticleSystem", "Graphics::GraphicsMesh",
            };
            bool risky = false;
            for (const char* k : kRisky) if (!strcmp(k, c.typeName)) risky = true;
            if (risky && !runRisky) {
                printf("%-28s SKIP (reaches an uninitialised subsystem; RISKY=1 to try) "
                       "-- A SKIP IS NOT A PASS\n", c.typeName);
                ++skipped; continue;
            }

            static unsigned char a[0x400], b[0x400];
            memset(a, 0xA5, sizeof(a));
            memset(b, 0x5A, sizeof(b));      // different poison on purpose

            static RtCtorStep tmp[512];
            // NEWSTEPS=1 runs the CreateNewObject table instead, compared
            // against the same DLL in-place factory: the two retail bodies
            // differ only by the allocation, so this verifies the OTHER table.
            const RtCtorStep* steps = getenv("NEWSTEPS") && c.newSteps
                                          ? c.newSteps : c.inPlaceSteps;
            if (neg) {
                int n = StepCount(steps);
                memcpy(tmp, steps, (n + 1) * sizeof(RtCtorStep));
                bool armed = false;
                for (int k = n - 1; k >= 0 && !armed; --k)
                    if (tmp[k].kind == RtFactOp::Imm) { tmp[k].imm ^= 1; armed = true; }
                if (!armed)                       // no immediate: corrupt a vptr
                    for (int k = n - 1; k >= 0 && !armed; --k)
                        if (tmp[k].kind == RtFactOp::Vptr) {
                            tmp[k].kind = RtFactOp::Imm; tmp[k].imm = 0; armed = true;
                        }
                if (!armed) printf("%-28s *** CONTROL COULD NOT BE ARMED\n", c.typeName);
                steps = tmp;
            }

            if (!getenv("THEIRSONLY") && !OursInPlace(steps, a, c.typeName)) { ++skipped; continue; }
            if (getenv("OURSONLY")) { printf("%-28s ours-side ran, no crash\n", c.typeName); continue; }
            theirs(b);

            int bad = 0, firstBad = -1;
            for (uint32_t o = 0; o < c.allocSize; ++o) {
                if (handle && o >= 0x08 && o < 0x10) continue;   // Internal handle id
                bool ourW = (a[o] != 0xA5), theirW = (b[o] != 0x5A);
                if (!ourW && !theirW) continue;
                if (ourW != theirW || a[o] != b[o]) { ++bad; if (firstBad < 0) firstBad = (int)o; }
            }
            if (deleg) { ++wrapChk; if (!bad) ++wrapPass; }
            else       { ++realChk; if (!bad) ++realPass; }
            if (!bad)
                printf("%-28s OK   (0x%x bytes)%s\n", c.typeName, c.allocSize,
                       deleg ? "   [WRAPPER ONLY: both sides call the same ctor]" : "");
            else
                printf("%-28s FAIL %d/%u bytes, first +0x%x (ours %02x theirs %02x)%s\n",
                       c.typeName, bad, c.allocSize, firstBad, a[firstBad], b[firstBad],
                       deleg ? "   [wrapper]" : "");
        }
    }

    printf("\n%s\n", neg ? "NEGATIVE CONTROL — every checked row MUST fail" : "ORACLE");
    printf("  shape (b) REAL store comparison : %d/%d\n", realPass, realChk);
    printf("  shape (c) WRAPPER ONLY          : %d/%d  (does NOT test the ctor)\n",
           wrapPass, wrapChk);
    printf("  shape (a) abstract null return  : %d/%d\n", stubPass, stubChk);
    printf("  skipped                         : %d   (a SKIP is not a pass)\n", skipped);
    return 0;
}
